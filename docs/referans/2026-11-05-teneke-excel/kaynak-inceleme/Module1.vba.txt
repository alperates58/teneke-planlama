Sub once_hedefleri_sqle_yaz()

    Dim tdSonRow As Long
    Dim tdSonColumn As Long
    Dim ctSonColumn As Long

    Dim pSeri As Long
    Dim pPlanaEklenenTarih As Long
    Dim pKisaKod As Long

    Dim pBitisTarih As Long

    Dim vSeri As String

    Dim vKisaKod As String

    Dim vBitisTarih As Variant
    Dim vUretimTarihi As Variant

    Dim vBaskiliStok As Double
    Dim vBaskiTalep As Double

Private Function SqlText(ByVal v As Variant) As String
    SqlText = Replace(Trim(CStr(v & "")), "'", "''")
End Function

Private Function SqlNum(ByVal v As Variant) As String
    If IsNumeric(v) Then
        SqlNum = Replace(CStr(CDbl(v)), ",", ".")
    Else
        SqlNum = "0"
    End If
End Function

Private Function SqlDate(ByVal v As Variant) As String
    If IsDate(v) Then
        SqlDate = "'" & Format(CDate(v), "yyyy-mm-dd hh:nn:ss") & "'"
    Else
        SqlDate = "NULL"
    End If
End Function

Private Function ToDouble(ByVal v As Variant) As Double
    If IsNumeric(v) Then
        ToDouble = CDbl(v)
    Else
        ToDouble = 0
    End If
End Function
